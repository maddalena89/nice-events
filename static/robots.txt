# The site is the home page plus a town page per commune (/in/<town>/) and a

# build time and all listed in the sitemap below.
#
# Copied verbatim into dist/ by niceevents/site.py, alongside sitemap.xml,
# which is generated on every build so its <lastmod> is the real build date.

User-agent: *
Allow: /

# Note on the /?e=<slug> URLs — deliberately NOT blocked.
#
# They serve the same page with one event highlighted, so they look like
# duplicates, and the tempting move is "Disallow: /*?e=". Don't. The structured
# data in the page points at exactly these URLs as the home of each event, and a
# blocked URL is one Google cannot read, cannot verify, and may drop the event
# for. Duplication is handled the correct way instead, by the <link rel=canonical>
# in the page, which tells Google these all consolidate to "/".
#
# The pretty short links (whatsonnice.com/<slug>) are a different thing: no file
# behind them, served by 404.html and bounced with JavaScript. Nothing on the
# site links to one, so nothing crawls into them. They exist for humans arriving
# from a QR code or a shared card.

Sitemap: https://whatsonnice.com/sitemap.xml

# AI crawlers. There are two kinds and they are NOT the same thing.
#
# SEARCH / ANSWER bots fetch a page so they can cite it in an answer, the way a
# search engine does. Blocking them does not protect anything — it just makes
# this site invisible to people asking "what's on in Nice this weekend" in
# ChatGPT or Perplexity. They are named explicitly below so nobody later
# "tidies up" the wildcard and silently removes the site from AI answers.

# block to uncomment is further down.
